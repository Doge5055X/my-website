const lerp =
    (a,b,t) =>
        a+(b-a)*t;

const easeOutCubic =
    t =>
        1-Math.pow(1-t,3);

const easeInCubic =
    t =>
        t*t*t;

const easeInOutCubic =
    t =>
        t < .5
            ? 4*t*t*t
            : 1-Math.pow(-2*t+2,3)/2;

const easeOutQuint =
    t =>
        1-Math.pow(1-t,5);

const easeInOutSine =
    t =>
        -(Math.cos(Math.PI*t)-1)/2;

const easeOutBack =
    t => {

        const c1 = 1.70158;
        const c3 = c1 + 1;

        return 1 +
            c3*Math.pow(t-1,3) +
            c1*Math.pow(t-1,2);
    };


/* ============================================================
   POINT

class Point {

    constructor(x,y) {

        this.x = x;
        this.y = y;
    }
}


/* ============================================================
   STROKE
   ============================================================

   Every final letter stroke has:

   • a final position
   • a central position
   • a phase
   • its own delay
   • its own personality

   The animation is therefore a geometric transformation,
   not a fade between two images.
   ============================================================ */

class MorphStroke {

    constructor(points, options={}) {

        this.points = points;

        this.delay =
            options.delay || 0;

        this.duration =
            options.duration || 700;

        this.phase =
            options.phase || 0;

        this.width =
            options.width || 5.6;

        this.opacity =
            options.opacity ?? 1;

        this.seedX =
            options.seedX ?? 500;

        this.seedY =
            options.seedY ?? 195;

        this.seedSpread =
            options.seedSpread ?? 0;

        this.seedAngle =
            options.seedAngle ?? 0;

        this.seedLength =
            options.seedLength ?? 90;

        this.index =
            options.index || 0;
    }


    /*
        Get a point's collapsed position.

        Every point goes to a location on the same vertical
        spine, but with tiny variations. This preserves the
        impression that the original geometry has been
        compressed rather than destroyed.
    */

    collapsedPoint(i) {

        const p =
            this.points[i];

        const center =
            this.seedY;

        const normalized =
            this.points.length <= 1
                ? .5
                : i/(this.points.length-1);

        /*
            Keep some of the original vertical information.
        */

        const vertical =
            (normalized-.5) *
            this.seedLength;

        /*
            Tiny lateral offsets give the collapsed object
            some internal texture.
        */

        const lateral =
            Math.sin(
                normalized *
                Math.PI *
                2 +
                this.phase
            ) *
            this.seedSpread;

        return new Point(
            this.seedX + lateral,
            center + vertical
        );
    }


    getPoint(i,time,mode) {

        const final =
            this.points[i];

        const collapsed =
            this.collapsedPoint(i);

        /*
            Collapse phase
        */

        if(mode === "collapse") {

            return {
                x: lerp(
                    final.x,
                    collapsed.x,
                    time
                ),

                y: lerp(
                    final.y,
                    collapsed.y,
                    time
                )
            };
        }


        /*
            Expansion phase.

            Use a little overshoot so the letters don't simply
            stop at their destination.
        */

        if(mode === "expand") {

            const overshoot =
                easeOutBack(time);

            /*
                The center is treated as the launch point.

                Slight curved deviation makes the expansion
                feel organic.
            */

            const dx =
                final.x -
                collapsed.x;

            const dy =
                final.y -
                collapsed.y;

            const perpendicularX =
                -dy;

            const perpendicularY =
                dx;

            const magnitude =
                Math.hypot(
                    perpendicularX,
                    perpendicularY
                ) || 1;

            const curve =
                Math.sin(
                    time *
                    Math.PI
                ) *
                7;

            return {
                x:
                    collapsed.x +
                    dx * overshoot +
                    perpendicularX/magnitude *
                    curve,

                y:
                    collapsed.y +
                    dy * overshoot +
                    perpendicularY/magnitude *
                    curve
            };
        }


        return final;
    }


    drawCollapse(progress) {

        ctx.beginPath();

        for(
            let i=0;
            i<this.points.length;
            i++
        ) {

            const p =
                this.getPoint(
                    i,
                    progress,
                    "collapse"
                );

            if(i===0)
                ctx.moveTo(
                    X(p.x),
                    Y(p.y)
                );
            else
                ctx.lineTo(
                    X(p.x),
                    Y(p.y)
                );
        }

        ctx.stroke();
    }


    drawExpand(progress) {

        ctx.beginPath();

        for(
            let i=0;
            i<this.points.length;
            i++
        ) {

            const p =
                this.getPoint(
                    i,
                    progress,
                    "expand"
                );

            if(i===0)
                ctx.moveTo(
                    X(p.x),
                    Y(p.y)
                );
            else
                ctx.lineTo(
                    X(p.x),
                    Y(p.y)
                );
        }

        ctx.stroke();
    }
}

/* ============================================================
   PARTICLE SYSTEM
   ============================================================ */

class Particle {

    constructor(x,y,energy=1) {

        this.x=x;
        this.y=y;

        const angle =
            Math.random() *
            Math.PI *
            2;

        const velocity =
            (.25+
            Math.random()*.9) *
            energy;

        this.vx =
            Math.cos(angle) *
            velocity;

        this.vy =
            Math.sin(angle) *
            velocity;

        this.life =
            300+
            Math.random()*900;

        this.maxLife =
            this.life;

        this.size =
            .35+
            Math.random()*1.4;
    }

    update(dt) {

        this.x +=
            this.vx *
            dt *
            .055;

        this.y +=
            this.vy *
            dt *
            .055;

        this.vx *= .985;
        this.vy *= .985;

        this.life -= dt;
    }

    draw() {

        const alpha =
            clamp(
                this.life /
                this.maxLife
            ) *
            .25;

        ctx.beginPath();

        ctx.arc(
            X(this.x),
            Y(this.y),
            X(this.size),
            0,
            Math.PI*2
        );

        ctx.fillStyle =
            `rgba(0,0,0,${alpha})`;

        ctx.fill();
    }
}

function emitCollapseParticles(progress) {

    /*
        Particles are pulled inward.

        Their positions are sampled from the letter geometry.
    */

    if(
        Math.random() >
        .22

        return;

    const stroke =
        strokes[
            Math.floor(
                Math.random()*
                strokes.length

    const point =
        stroke.points[
            Math.floor(
                Math.random()*
                stroke.points.length
            )

    particles.push(
        new Particle(
            point.x,
            point.y,
            .35
        )
    );
}

/* ============================================================
   CENTER SPINE
   ============================================================ */

function drawSpine(progress,mode) {

    /*
        This is not simply a new line.

        It represents the mass of the collapsed letters.
    */

    let alpha=0;

    if(mode==="collapse")
        alpha=
            easeInOutSine(progress);

    if(mode==="expand")
        alpha=
            1-easeOutCubic(progress);

    if(alpha<=0)
        return;

    ctx.save();

    ctx.lineCap="round";

    /*
        Outer glow
    */

    ctx.shadowColor=
        `rgba(0,0,0,${alpha*.22})`;

    ctx.shadowBlur=
        X(12);

    ctx.strokeStyle=
        `rgba(15,15,15,${alpha*.12})`;

    ctx.lineWidth=
        X(13);

    ctx.beginPath();

    ctx.moveTo(
        X(500),
        Y(125)
    );

    ctx.lineTo(
        X(500),
        Y(265)
    );

    ctx.stroke();


    /*
        Core
    */

    ctx.shadowBlur=0;

    ctx.strokeStyle=
        `rgba(12,12,12,${alpha*.8})`;

    ctx.lineWidth=
        X(3.2);

    ctx.beginPath();

    ctx.moveTo(
        X(500),
        Y(125)
    );

    ctx.lineTo(
        X(500),
        Y(265)
    );

    ctx.stroke();

    ctx.restore();
}

   FINAL WORD
   ============================================================ */

function drawInitialWord() {

    ctx.save();

    ctx.lineCap="round";
    ctx.lineJoin="round";

    ctx.lineWidth=
        X(5.6);

    ctx.strokeStyle=
        "#111";

    for(const stroke of strokes) {

        ctx.beginPath();

        for(
            let i=0;
            i<stroke.points.length;
            i++
        ) {

            const p =
                stroke.points[i];

            if(i===0)
                ctx.moveTo(
                    X(p.x),
                    Y(p.y)
                );
            else
                ctx.lineTo(
                    X(p.x),
                    Y(p.y)
                );
        }

        ctx.stroke();
    }

    ctx.restore();
}


/* ============================================================
   COLLAPSE
   ============================================================ */

function drawCollapse(progress) {

    const eased =
        easeInOutCubic(progress);

    ctx.save();

    ctx.lineCap="round";
    ctx.lineJoin="round";

    ctx.lineWidth=
        X(5.6);

    ctx.strokeStyle=
        "#111";

    /*
        All strokes collapse simultaneously.

        But not at exactly the same speed.
    */

    for(const stroke of strokes) {

        const local =
            clamp(
                (eased -
                stroke.index*.008) /
                .94
            );

        stroke.drawCollapse(
            easeInOutCubic(local)
        );
    }

    ctx.restore();

    drawSpine(
        progress,
        "collapse"
    );

    emitCollapseParticles(
        progress
    );
}


/* ============================================================
   EXPANSION
   ============================================================ */

function drawExpansion(progress) {

    /*
        The expansion is a wave.

        Each stroke begins within a few milliseconds of
        its neighbours, creating the illusion that the
        central bar is exploding into the word.
    */

    ctx.save();

    ctx.lineCap="round";
    ctx.lineJoin="round";

    ctx.lineWidth=
        X(5.6);

    ctx.strokeStyle=
        "#111";

    for(const stroke of strokes) {

        /*
            Extremely tight staggering.
        */

        const stagger =
            stroke.index *
            .018;

        const local =
            clamp(
                (progress-stagger) /
                .82
            );

        const eased =
            easeOutBack(local);

        stroke.drawExpand(
            eased
        );
    }

    ctx.restore();

    drawSpine(
        progress,
        "expand"
    );

    emitExpansionParticles(
        progress
    );
}


/* ============================================================
   FINAL SETTLE
   ============================================================ */

function drawFinal(time) {

    /*
        Very subtle breathing shadow.

        The geometry itself stays perfectly still.
    */

    const breathing =
        0.018+
        Math.sin(time*.002)*.008;

    ctx.save();

    ctx.shadowColor=
        `rgba(0,0,0,${breathing})`;

    ctx.shadowBlur=
        X(8);

    ctx.lineCap="round";
    ctx.lineJoin="round";

    ctx.lineWidth=
        X(5.6);

    ctx.strokeStyle=
        "#111";

    for(const stroke of strokes) {

        ctx.beginPath();

        for(
            let i=0;
            i<stroke.points.length;
            i++
        ) {

            const p =
                stroke.points[i];

            if(i===0)
                ctx.moveTo(
                    X(p.x),
                    Y(p.y)
                );
            else
                ctx.lineTo(
                    X(p.x),
                    Y(p.y)
                );
        }

        ctx.stroke();
    }

    ctx.restore();
}


/* ============================================================

let running=false;
let startTime=0;
let last=0;

function render(now) {

    const elapsed =
        now-startTime;

    const dt =
        Math.min(
            now-last,
            32
        );

    last=now;

    background();


    /* -----------------------------------------------
       PHASE 1 — WORD EXISTS
       ----------------------------------------------- */

    if(
        elapsed <
        TIMELINE.intro
    ) {

        drawInitialWord();
    }


    /* -----------------------------------------------
       PHASE 2 — COLLAPSE
       ----------------------------------------------- */

    else if(
        elapsed <
        TIMELINE.intro +
        TIMELINE.collapse
    ) {

        const t =
            (elapsed-
            TIMELINE.intro) /
            TIMELINE.collapse;

        drawCollapse(
            clamp(t)
        );
    }


    /* -----------------------------------------------
       PHASE 3 — THE SINGULARITY
       ----------------------------------------------- */

    else if(
        elapsed <
        TIMELINE.intro +
        TIMELINE.collapse +
        TIMELINE.hold
    ) {

        /*
            Keep the bar absolutely still.

            This tiny pause is important.

            The brain gets a fraction of a second to register
            that everything has become one object.
        */

        drawSpine(
            1,
            "collapse"
        );
    }


    /* -----------------------------------------------
       PHASE 4 — EXPANSION
       ----------------------------------------------- */

    else if(
        elapsed <
        TIMELINE.intro +
        TIMELINE.collapse +
        TIMELINE.hold +
        TIMELINE.expand
    ) {

        const beginning =
            TIMELINE.intro +
            TIMELINE.collapse +
            TIMELINE.hold;

        const t =
            (elapsed-beginning) /
            TIMELINE.expand;

        drawExpansion(
            clamp(t)
        );
    }


    /* -----------------------------------------------
       PHASE 5 — FINAL STATE
       ----------------------------------------------- */

    else {

        drawFinal(elapsed);

        /*
            A tiny final shimmer of particles.
        */

        if(
            Math.random()<.015
        ) {

            particles.push(
                new Particle(
                    500+
                    (Math.random()-.5)*300,

                    195+
                    (Math.random()-.5)*80,

                    .1
                )
            );
        }
    }


    /* -----------------------------------------------
       PARTICLES
       ----------------------------------------------- */

    for(
        let i=particles.length-1;
        i>=0;
        i--
    ) {

        const p=
            particles[i];

        p.update(dt);
        p.draw();

        if(
            p.life<=0
        ) {

            particles.splice(
                i,
                1
            );
        }
    }


    if(
        elapsed <
        TOTAL+1000
    ) {

        requestAnimationFrame(
            render
        );

    } else {

        running=false;
    }
}
